\noindent Обоснование: ФИО — VARCHAR(100). Согласно Приказу ФМС России \\ от 30.06.2011 №~279 «Об утверждении Правил и способа формирования машиночитаемой записи в паспорте гражданина Российской Федерации», верхняя строка машиночитаемой записи отводит под ФИО 39 символов — это аппаратное ограничение физического бланка паспорта. Поскольку в системе ФИО хранится одной строкой в кириллице, а не в транслитерации, реальная длина оказывается больше: с учётом двойных фамилий и длинных отчеств она может достигать 60--70 символов. Выбрано 100 с запасом. Логин — VARCHAR(30). Логин является произвольной строкой для входа в систему. Выбрано 30 как разумный предел, исключающий избыточно длинные значения.

\subsection{Лингвистическое описание схемы базы данных}

\begin{enumerate}
    \item Тип сотрудника (тип\_сотрудника\_id(SERIAL), название\_типа(VARCHAR(30)), описание(TEXT)).
    \item Статус сотрудника (статус\_id(SERIAL), название(VARCHAR(20))).
    \item Сотрудник (сотрудник\_id(SERIAL), тип\_сотрудника\_id(INT), фио(VARCHAR(100)), логин(VARCHAR(30)), статус\_id(INT)).

    \item Тип зоны парковки (тип\_зоны\_id(SERIAL), название(VARCHAR(20))).
    \item Статус зоны парковки (статус\_зоны\_id(SERIAL), название(VARCHAR(25))).
    \item Зона парковки (зона\_парковки\_id(SERIAL), название(VARCHAR(60)), тип\_зоны\_id(INT), район\_города(VARCHAR(50)), макс\_авто(INT), статус\_зоны\_id(INT)).
    \item Координаты зоны парковки (координаты\_зоны\_id(SERIAL), зона\_парковки\_id(INT), номер\_вершины(INT), широта(NUMERIC(8, 5)), долгота(NUMERIC(8, 5))).

    \item Статус автомобиля (статус\_id(SERIAL), название(VARCHAR(25))).
    \item Автомобиль (авто\_id(SERIAL), гос\_номер(VARCHAR(9)), марка(VARCHAR(50)), модель(VARCHAR(70)), год\_выпуска(INT), vin\_номер(VARCHAR(17)), статус\_id(INT)).
    \item Парковочная сессия (сессия\_id(SERIAL), авто\_id(INT), зона\_парковки\_id(INT), время\_въезда(TIMESTAMP), время\_выезда(TIMESTAMP)).

    \item Тип статуса трека (статус\_трека\_id(SERIAL), название\_типа(VARCHAR(20))).
    \item Тип вида трека (вид\_трека\_id(SERIAL), название\_типа(VARCHAR(25))).
    \item Трек движения (трек\_id(SERIAL), авто\_id(INT), время\_начала(TIMESTAMP), время\_конца(TIMESTAMP), статус\_трека\_id(INT), вид\_трека\_id(INT)).
    \item Источник данных (источник\_id(SERIAL), название(VARCHAR(25))).
    \item Координаты точки трека (id\_точки(SERIAL), id\_трека(INT), id\_авто(INT), широта(NUMERIC(8, 5)), долгота(NUMERIC(8, 5)), скорость(NUMERIC(5, 2), источник\_id(INT)).

    \item Тип тревожного события (тип\_события\_id(SERIAL), название\_типа(VARCHAR(25)), описание(TEXT)).
    \item Статус обработки события (статус\_id(SERIAL), название(VARCHAR(20))).
    \item Тревожное событие (событие\_id(SERIAL), авто\_id(INT), сотрудник\_id(INT), тип\_события\_id(INT), широта(NUMERIC(8, 5)), долгота(NUMERIC(8, 5)), описание(TEXT), статус\_id(INT)).

    \item Тип обращения к геомониторингу (тип\_запрос\_id(SERIAL), название\_типа(VARCHAR(25)), описание(TEXT)).
    \item Обращение к геомониторингу (запрос\_id(SERIAL), сотрудник\_id(INT), авто\_id(INT), тип\_запроса\_id(INT), цель\_запроса(TEXT)).
\end{enumerate}
