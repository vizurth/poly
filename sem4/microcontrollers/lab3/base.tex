\section{Цель работы}
Ознакомится с основными методами обработки событий и прерываний.

\section{Постановка задачи}
разработать программу для микроконтроллера (МК)
STM32F200 мигающую светодиодом PG7 и регистрирующую и обрабатывающую с
разным приоритетом замыкание кнопок “WAKEUP” и “USER”. При нажатии
кнопки “WAKEUP” на некоторое время должен подключаться светодиод PG6, а при
нажатии кнопки “USER” светодиод PG8 (см. рис. 1).

\section{Алгоритм работы программы}

\subsection{Инициализация проекта}
\begin{itemize}
    \item Создание нового проекта в среде Keil $\mu$Vision5.
    \item Выбор микроконтроллера \textbf{STM32F207IGHx}.
    \item Настройка параметров проекта, включая поддержку ядра (\textit{CMSIS CORE}), запуск (\textit{Startup}) и настройки HAL.
    \item Подключение библиотеки \texttt{stm32f2xx.h}, где хранится информация обо всех адресах регистров.
\end{itemize}

\subsection{Настройка портов GPIOG и GPIOA}
\begin{itemize}
    \item Включение тактирования порта GPIOG через регистр \texttt{RCC->AHB1ENR}.
    \item Настройка пинов PG6, PG7, PG8 как выходы через \texttt{GPIOG->MODER}.
    \item Установка типа выхода в тяни-толкай (push-pull) через \texttt{GPIOG->OTYPER}.
    \item Установка низкой скорости переключения для порта GPIOG через \texttt{GPIOG->OSPEEDR}.
    \item Настройка всех пинов GPIOA как входы через \texttt{GPIOA->MODER}.
    \item Аналогичная настройка типа выхода и скорости переключения для GPIOA.
\end{itemize}

\subsection{Настройка внешних прерываний (EXTI)}
\begin{itemize}
    \item Включение тактирования контроллера конфигурации системы (SYSCFG) через \texttt{RCC->APB2ENR}.
    \item Разрешение прерываний для линий EXTI0 (PA0) и EXTI15 (PG15) через \texttt{EXTI->IMR}.
    \item Настройка прерывания EXTI0 по нарастающему фронту ($0 \to 1$) через \texttt{EXTI->RTSR}.
    \item Настройка прерывания EXTI15 по спадающему фронту ($1 \to 0$) через \texttt{EXTI->FTSR}.
    \item Связывание EXTI0 с PA0, а EXTI15 с PG15 через \texttt{SYSCFG->EXTICR}.
\end{itemize}

\subsection{Настройка и активация прерываний в NVIC}
\begin{itemize}
    \item Установка групповых приоритетов прерываний через \texttt{NVIC\_SetPriorityGrouping}.
    \item Установка приоритетов для EXTI0 и EXTI15\_10.
    \item Активация обработчиков прерываний EXTI0 и EXTI15\_10 через \texttt{NVIC\_EnableIRQ}.
\end{itemize}

\subsection{Реализация мигания светодиодов}
\begin{enumerate}
    \item \textbf{Основной цикл программы:}
    \begin{itemize}
        \item Постоянное мигание PG7 с задержкой.
    \end{itemize}
    
    \item \textbf{Обработчик прерывания EXTI0} (нажатие на PA0):
    \begin{itemize}
        \item 10-кратное мигание PG6.
        \item Очистка флага прерывания EXTI0.
    \end{itemize}
    
    \item \textbf{Обработчик прерывания EXTI15} (нажатие на PG15):
    \begin{itemize}
        \item 10-кратное мигание PG8.
        \item Очистка флага прерывания EXTI15.
    \end{itemize}
\end{enumerate}

\subsection{Используемые переменные и регистры}
\begin{description}
    \item[\texttt{RCC->AHB1ENR}] — включает тактирование GPIOG.
    \item[\texttt{RCC->APB2ENR}] — включает тактирование контроллера конфигурации (SYSCFG).
    \item[\texttt{GPIOG->MODER}] — задает режим работы пинов PG6, PG7, PG8 (выход).
    \item[\texttt{GPIOG->OTYPER}] — определяет тип выхода (push-pull).
    \item[\texttt{GPIOG->OSPEEDR}] — определяет скорость переключения (низкая).
    \item[\texttt{GPIOA->MODER}] — все пины настроены как входы.
    \item[\texttt{EXTI->IMR}] — маска разрешенных внешних прерываний (EXTI0, EXTI15).
    \item[\texttt{EXTI->RTSR}] — определяет срабатывание EXTI0 по нарастающему фронту.
    \item[\texttt{EXTI->FTSR}] — определяет срабатывание EXTI15 по спадающему фронту.
    \item[\texttt{SYSCFG->EXTICR}] — связывает EXTI0 с PA0, а EXTI15 с PG15.
    \item[\texttt{NVIC\_SetPriorityGrouping}] — устанавливает групповой приоритет прерываний.
    \item[\texttt{NVIC\_SetPriority}] — задает приоритет конкретных прерываний.
    \item[\texttt{NVIC\_EnableIRQ}] — включает прерывания EXTI0 и EXTI15\_10.
    \item[\texttt{GPIOG->ODR}] — управляет состоянием светодиодов.
    \item[\texttt{EXTI->PR}] — очищает флаг прерывания после его обработки.
\end{description}

\section*{Код программы}

\begin{lstlisting}[style=cstyle, caption={Основная функция программы}]
#include <stm32f207xx.h>
#include <stm32f2xx.h>
#include <core_cm3.h>

// Функция задержки
// Просто крутит пустой цикл, чтобы процессор был чем-то занят
// Чем больше число t, тем дольше ждать
void delay (unsigned long t)
{
    unsigned long i;
    for (i = 0; i < t; ++i);
}

int main (void)
{
    // Включаем тактирование
    // Чтобы порты G и A заработали, на них нужно подать тактовый сигнал
    // Бит 6 включает порт G, Бит 0 включает порт A
    RCC->AHB1ENR |= (RCC_AHB1ENR_GPIOGEN | RCC_AHB1ENR_GPIOAEN); 

    // Настройка светодиодов на порту G
    // Настраиваем пины 6, 7 и 8 порта G как выходы
    // Это нужно, чтобы мы могли подавать на них напряжение и зажигать диоды
    GPIOG->MODER |= (GPIO_MODER_MODE6_0 | GPIO_MODER_MODE7_0 | GPIO_MODER_MODE8_0);
    
    // Настраиваем тип выхода как обычный
    GPIOG->OTYPER = 0; 
    
    // Скорость переключения пинов ставим низкую
    // Для мигания светодиодом высокая скорость не нужна
    GPIOG->OSPEEDR = 0; 

    // Настройка кнопки на порту A
    // Настраиваем весь порт А на вход
    // Микроконтроллер будет слушать напряжение на ножке
    GPIOA->MODER = 0; 
    GPIOA->OTYPER = 0;
    GPIOA->OSPEEDR = 0;

    // Включаем тактирование системного контроллера SYSCFG
    // Он нужен, чтобы подключить ножку порта к линии прерывания
    RCC->APB2ENR |= RCC_APB2ENR_SYSCFGEN; 

    // Связываем ножки с линиями прерываний
    // Линия 0 будет слушать порт A, это пин PA0
    SYSCFG->EXTICR[0] |= SYSCFG_EXTICR1_EXTI0_PA; 
    
    // Линия 15 будет слушать порт G, это пин PG15
    SYSCFG->EXTICR[3] |= SYSCFG_EXTICR4_EXTI15_PG; 

    // Настройка самих прерываний EXTI
    // Разрешаем прерывания на линиях 0 и 15
    // Если этого не сделать, процессор будет игнорировать сигналы
    EXTI->IMR |= (EXTI_IMR_MR0 | EXTI_IMR_MR15);
    
    // Настраиваем срабатывание по переднему фронту для линии 0
    // Прерывание сработает, когда сигнал пойдет вверх
    EXTI->RTSR |= EXTI_RTSR_TR0; 
    
    // Настраиваем срабатывание по заднему фронту для линии 15
    // Прерывание сработает, когда сигнал пойдет вниз
    EXTI->FTSR |= EXTI_FTSR_TR15; 

    // Настройка приоритетов
    // Задаем группировку приоритетов, это схема разделения важности задач
    NVIC_SetPriorityGrouping (0x5UL); 

    // Устанавливаем важность для прерывания кнопки PA0
    // Чем меньше число, тем важнее прерывание
    NVIC_SetPriority (EXTI0_IRQn, 2 | 2 << 2); 
    
    // Устанавливаем важность для прерывания кнопки PG15
    // Это прерывание будет важнее предыдущего
    NVIC_SetPriority (EXTI15_10_IRQn, 1 | 1 << 2);

    // Активируем эти прерывания в самом процессоре
    // Теперь процессор готов реагировать на события
    NVIC_EnableIRQ (EXTI0_IRQn); 
    NVIC_EnableIRQ (EXTI15_10_IRQn);

    // Главный цикл программы
    // Пока никто не нажимает кнопки, про��ессор просто мигает диодом на пине 7
    for (;;) {
        // Меняем состояние пина 7 на противоположное
        GPIOG->ODR ^= GPIO_ODR_OD7;
        delay (200000); 
    }
}

// Обработчик прерывания для линии 0, когда нажали PA0
// Эта функция вызывается сама аппаратно
void EXTI0_IRQHandler (void)
{
    unsigned int i;
    // Мигаем другим диодом PG6 быстро 60 раз
    for (i = 0; i < 60; ++i) {
        GPIOG->ODR ^= GPIO_ODR_OD6; 
        delay (100000);
    }
    
    // Сбрасываем флаг прерывания
    // Мы пишем 1 в бит PR0, чтобы подтвердить выполнение
    EXTI->PR |= EXTI_PR_PR0; 
}

// Обработчик прерывания для линий 10-15, когда сработал PG15
// Эта функция вызывается сама аппаратно
void EXTI15_10_IRQHandler (void)
{
    unsigned int i;
    // Мигаем третьим диодом PG8 медленно 30 раз
    for (i = 0; i < 30; ++i) {
        GPIOG->ODR ^= GPIO_ODR_OD8; 
        delay (300000);
    }
    
    // Сбрасываем флаг прерывания для линии 15
    // Пишем 1 в бит PR15, подтверждая выполнение
    EXTI->PR |= EXTI_PR_PR15; 
}
\end{lstlisting}
